13NAV::SatelliteSystem NAV::vendor::vectornav::toSatelliteSystem(NAV::vendor::vectornav::SatSys sys)
14{
15 SatelliteSystem satSys = SatSys_None;
16 switch (sys)
17 {
18 case vendor::vectornav::SatSys::GPS:
19 satSys = GPS;
20 break;
21 case vendor::vectornav::SatSys::SBAS:
22 satSys = SBAS;
23 break;
24 case vendor::vectornav::SatSys::Galileo:
25 satSys = GAL;
26 break;
27 case vendor::vectornav::SatSys::BeiDou:
28 satSys = BDS;
29 break;
30 case vendor::vectornav::SatSys::IMES:
31 LOG_TRACE("VectorNav SatInfoElement satellite system '{}' is not supported yet. Skipping measurement.", sys);
32 break;
33 case vendor::vectornav::SatSys::QZSS:
34 satSys = QZSS;
35 break;
36 case vendor::vectornav::SatSys::GLONASS:
37 satSys = GLO;
38 break;
39 default: // IRNSS not in vectorNav
40 LOG_TRACE("VectorNav SatInfoElement satellite system '{}' is not supported yet. Skipping measurement.", sys);
41 break;
42 }
43 return satSys;
44}

46NAV::vendor::vectornav::SatSys NAV::vendor::vectornav::fromSatelliteSystem(NAV::SatelliteSystem sys)
47{
48 switch (sys.toEnumeration())
49 {
50 case SatelliteSystem::Enum_GPS:
51 return vendor::vectornav::SatSys::GPS;
52 case SatelliteSystem::Enum_GAL:
53 return vendor::vectornav::SatSys::Galileo;
54 case SatelliteSystem::Enum_GLO:
55 return vendor::vectornav::SatSys::GLONASS;
56 case SatelliteSystem::Enum_BDS:
57 return vendor::vectornav::SatSys::BeiDou;
58 case SatelliteSystem::Enum_QZSS:
59 return vendor::vectornav::SatSys::QZSS;
60 case SatelliteSystem::Enum_SBAS:
61 return vendor::vectornav::SatSys::SBAS;
62 case SatelliteSystem::Enum_IRNSS:
63 case SatelliteSystem::Enum_COUNT:
64 case SatelliteSystem::Enum_None:
65 LOG_TRACE("'{}' is not supported in the VectorNav satellite systems. Defaulting to GPS.", sys);
66 return vendor::vectornav::SatSys::GPS;
67 }
68 return vendor::vectornav::SatSys::GPS;
