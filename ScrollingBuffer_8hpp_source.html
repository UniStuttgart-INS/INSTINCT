24/// @brief A buffer which is overwriting itself from the start when full
25/// @tparam T Type of data stored in the buffer
26/// @tparam _Padding The padding are empty values at the start of the buffer to prevent overriding the start value in multithreaded applications
27template<class T, size_t _Padding = 0>
28class ScrollingBuffer

35 /// @brief Reserves space for the buffer but does not fill the buffer with values
36 /// @param[in] maxSize The maximum size of the scrolling buffer
37 explicit ScrollingBuffer(size_t maxSize = 0)
38 : _infiniteBuffer(maxSize == 0), _maxSize(maxSize + _Padding), _dataStart(_Padding), _dataEnd(_infiniteBuffer ? 0 : _Padding)
39 {
40 _data.reserve(maxSize + _Padding);
41 _data.resize(_Padding);

44 /// @brief Constructs a new container with the contents of the initializer list init.
45 /// @param[in] init initializer list to initialize the elements of the container with
46 ScrollingBuffer(std::initializer_list<T> init)
47 : _infiniteBuffer(init.size() == 0), _maxSize(init.size())
48 {
49 _data.reserve(init.size());
50
51 for (auto& val : init)
52 {
53 push_back(val);
54 }
55 }

149 /// @param[in] index Iterator index inside the buffer (counted from the start)
150 explicit Iterator(ScrollingBuffer& buffer, size_t index = 0)

407 /// @brief Checks if the container has no elements
408 [[nodiscard]] bool empty() const
409 {
410 return size() == 0;
411 }
412
413 /// @brief Checks if the container is full (never full when infinite buffer)
414 [[nodiscard]] bool full() const
415 {
416 return !_infiniteBuffer && _maxSize - _Padding == size();
417 }
418
419 /// @brief Returns the number of elements in the container
420 [[nodiscard]] size_t size() const
421 {
422 if (_dataStart == _Padding && _dataEnd == _Padding) // Buffer empty or full and not scrolled
423 {
424 return _data.size() - _Padding;
425 }
426 if (_dataStart < _dataEnd) // not scrolled buffer
427 {
428 return _dataEnd - _dataStart;
429 }
430
431 // scrolled buffer
432 return _maxSize - (_dataStart - _dataEnd);
433 }

452 /// @brief Erases all elements from the container. After this call, size() returns zero.
453 void clear()
454 {
455 _data.clear();
456 _dataStart = _Padding;
457 _dataEnd = _infiniteBuffer ? 0 : _Padding;
458
459 for (size_t i = 0; i < _Padding; i++)
460 {
461 _data.push_back({});
462 }

465 /// @brief Appends the given element value to the end of the container.
466 /// @param[in] value the value of the element to append
467 void push_back(const T& value)
468 {
469 if (_infiniteBuffer) // The buffer should grow when adding new values
470 {
471 _data.push_back(value);
472 _maxSize = _data.size();
473 }
474 else if (_data.size() < _maxSize) // The real buffer is smaller than the allowed buffer size
475 {
476 _data.push_back(value);
477 _dataEnd = (_dataEnd + 1) % _maxSize;
478 }
479 else // The real buffer as large as or bigger than the allowed buffer size, so we have to scroll the buffer
480 {
481 _data.at(_dataEnd) = value;
482 if (size() >= capacity())
483 {
484 // "5, 6, _, _, 2, 3, 4"
485 // "5, 6, 7, _, 2, 3, 4"
486 _dataStart = (_dataStart + 1) % _maxSize;
487 }
488 _dataEnd = (_dataEnd + 1) % _maxSize;
489 }
490 }

492 /// @brief Removes the first element of the container
493 void pop_front()
494 {
495 if (empty())
496 {
497 return;
498 }
499 if (size() == 1)
500 {
501 clear();
502 return;
503 }
504
505 if (_infiniteBuffer)
506 {
507 _data.erase(_data.begin());
508 _maxSize = _data.size();
509 }
510 else
511 {
512 // se e s e s s e se
513 // 5, 6, 2, 3, 4 // 5, 6, _, _, 2, 3, 4 // 5, 6, X, X, 2, 3, 4 // X, 6, 7, 8, 9, 10, X // 5, 6, 7, 4
514 // e s e s e s s e s e
515 // 5, 6, _, 3, 4 // 5, 6, _, _, _, 3, 4 // 5, 6, _, X, X, 3, 4 // X, X, 7, 8, 9, 10, _ // 5, 6, 7, _
516
517 front() = {}; // Set to default in case shared_ptr is stored
518 _dataStart = (_dataStart + 1) % _maxSize;
519 }
520 }

522 /// @brief Removes the last element of the container
523 void pop_back()
524 {
525 if (empty())
526 {
527 return;
528 }
529 if (size() == 1)
530 {
531 clear();
532 return;
533 }
534
535 if (_infiniteBuffer)
536 {
537 _data.pop_back();
538 _maxSize = _data.size();
539 }
540 else
541 {
542 back() = {}; // Set to default in case shared_ptr is stored
543 _dataEnd = _dataEnd == 0 ? _maxSize - 1 : (_dataEnd - 1);
544 }
545 }

547 /// @brief Resizes the buffer to the specified size
548 /// @param[in] targetSize The new buffer size (0 for infinite buffer)
549 void resize(size_t targetSize)
550 {
551 if (targetSize == 0) // Buffer should grow indefinitely when adding new values
552 {
553 _infiniteBuffer = true;
554
555 if (isScrolled()) // Buffer is scrolled and needs to be sorted in order of insertion
556 {
557 // se e s e s s e
558 // 5, 6, 2, 3, 4 // 5, 6, _, _, 2, 3, 4 // 5, 6, X, X, 2, 3, 4 // X, 6, 7, 8, 9, 10, X
559 // 2, 3, 4, 5, 6 // 2, 3, 4, 5, 6, _, _ // X, X, 2, 3, 4, 5, 6 // X, X, 6, 7, 8, 9, 10
560 std::vector<T> to_vector;
561
562 if (_dataEnd > _dataStart)
563 {
564 std::copy(std::next(_data.begin(), static_cast<int64_t>(_dataEnd)), _data.end(),
565 std::back_inserter(to_vector));
566 _maxSize = _dataEnd;
567 }
568
569 std::copy(std::next(_data.begin(), std::max(static_cast<int>(_dataStart - _Padding), 0)),
570 std::next(_data.begin(), static_cast<int64_t>(std::max(_dataEnd, _maxSize))),
571 std::back_inserter(to_vector));
572
573 if (int64_t elementsFront = std::min(static_cast<int64_t>(_dataEnd), static_cast<int64_t>(_dataStart - _Padding));
574 elementsFront > 0)
575 {
576 std::copy(_data.begin(), std::next(_data.begin(), elementsFront),
577 std::back_inserter(to_vector));
578 }
579 _data.swap(to_vector);
580
581 _maxSize = _data.size();
582
583 _dataStart = _Padding;
584 _dataEnd = 0;
585 }
586 }
587 else // Buffer should have scrolling behaviour when adding new values
588 {
589 _infiniteBuffer = false;
590
591 if (_maxSize - _Padding > targetSize) // We make the buffer smaller
592 {
593 if (!isScrolled()) // Buffer is not scrolled, so shrinking removes the values from the front of the buffer
594 {
595 size_t elementsToDelete = _maxSize - _Padding - targetSize;
596
597 if (size_t emptyAtTheBack = _maxSize - _dataEnd;
598 size_t emptyAtTheBackToDelete = std::min(emptyAtTheBack, elementsToDelete))
599 {
600 // 1, 2, 3, _, _, // X, X, 1, 2, 3, _, _,
601 // 1, 2, 3, _, // X, X, 1, 2, 3, _,
602 _maxSize -= emptyAtTheBackToDelete;
603 _dataEnd %= _maxSize; // NOLINT(clang-analyzer-core.DivideZero) // this lint is wrong, even when wrapped by `if(_maxSize != 0)` appearing
604 elementsToDelete -= emptyAtTheBackToDelete;
605 // 1, 2, 3, // X, X, 1, 2, 3,
606 }
607
608 // 1, 2, 3,
609 // 2, 3,
610 if (elementsToDelete)
611 {
612 _data.erase(std::next(_data.begin(), static_cast<int64_t>(_dataStart)),
613 std::next(_data.begin(), static_cast<int64_t>(_dataStart + elementsToDelete)));
614 _maxSize -= elementsToDelete;
615 }
616 }
617 else // Buffer is scrolled, so the correct values have to be erased from the buffer when shrinking
618 {
619 // 5, 6, _, _, 2, 3, 4, // 5, 6, _, _, X, X, 2, 3, 4,
620 // 6, // 6, X, X
621
622 // 6, 7, 3, 4, 5, // 5, 6, X, X, 2, 3, 4
623 // 6, 7, 4, 5, // 5, 6, X, X, 3, 4
624
625 // X, 6, 7, 8, 9, 10, X
626 // X, 9, 10, X
627
628 size_t elementsToDelete = _maxSize - _Padding - targetSize;
629
630 if (size_t emptyInBetween = static_cast<size_t>(std::max(static_cast<int>(_dataStart - _Padding - _dataEnd), 0));
631 size_t emptyInBetweenToDelete = std::min(emptyInBetween, elementsToDelete))
632 {
633 // 5, 6, _, _, 2, 3, 4, // 5, 6, _, _, X, X, 2, 3, 4,
634 _data.erase(std::next(_data.begin(), static_cast<int64_t>(_dataEnd)),
635 std::next(_data.begin(), static_cast<int64_t>(_dataEnd + emptyInBetweenToDelete)));
636 // 5, 6, _, 2, 3, 4, // 5, 6, _, X, X, 2, 3, 4,
637 _dataStart -= emptyInBetweenToDelete;
638 _maxSize -= emptyInBetweenToDelete;
639 elementsToDelete -= emptyInBetweenToDelete;
640 }
641
642 // s e
643 // X , 6, 7 , 8 , 9, 10, X
644 // X(8), 9, 10, X(7)
645 if (size_t paddingAtTheEnd = static_cast<size_t>(std::max(static_cast<int>(_Padding - _dataStart), 0));
646 size_t paddingAtTheEndToDelete = std::min(paddingAtTheEnd, elementsToDelete))
647 {
648 _data.erase(std::next(_data.begin(), static_cast<int64_t>(_dataEnd)),
649 std::next(_data.begin(), static_cast<int64_t>(_dataEnd + paddingAtTheEndToDelete)));
650 _maxSize -= paddingAtTheEndToDelete;
651 _dataStart += paddingAtTheEndToDelete;
652 _dataEnd %= _maxSize;
653 elementsToDelete -= paddingAtTheEndToDelete;
654 }
655 // e s
656 // X, X, 6, 7 , 8 , 9, 10
657 // X, X, 7 , 8 , 9, 10
658
659 if (size_t elementsAtTheBack = _maxSize - _dataStart - (_dataEnd > _dataStart ? _maxSize - _dataEnd : 0);
660 size_t elementsAtTheBackToDelete = std::min(elementsAtTheBack, elementsToDelete))
661 {
662 // 5, 6, 2, 3, 4, // 5, 6, X, X, 2, 3, 4,
663 _data.erase(std::next(_data.begin(), static_cast<int64_t>(_dataStart - _Padding)),
664 std::next(_data.begin(), static_cast<int64_t>(_dataStart - _Padding + elementsAtTheBackToDelete)));
665 // 5, 6, 4, // 5, 6, X, X, 4,
666 // 5, 6, // 5, 6, X, X,
667 _maxSize -= elementsAtTheBackToDelete;
668 _dataStart %= _maxSize;
669 if (_dataEnd > _maxSize)
670 {
671 _dataEnd -= elementsToDelete;
672 }
673 _dataEnd %= _maxSize;
674 elementsToDelete -= elementsAtTheBackToDelete;
675 }
676 if (elementsToDelete)
677 {
678 // 5, 6, // 5, 6, X, X,
679 _data.erase(std::next(_data.begin(), static_cast<int64_t>(_dataStart)),
680 std::next(_data.begin(), static_cast<int64_t>(std::min(_dataStart + elementsToDelete, _maxSize - _Padding))));
681 // 6, // 6, X, X
682 _maxSize -= elementsToDelete;
683 if (_dataEnd >= elementsToDelete)
684 {
685 _dataEnd -= elementsToDelete;
686 }
687 }
688 }
689 }
690 else if (_maxSize - _Padding < targetSize) // We make the buffer bigger
691 {
692 // 1, 2, 3, _, _, // X, X, 0, 1, 2, 3, _, _
693 // 1, 2, 3, _, _, _, _, // X, X, 0, 1, 2, 3, _, _, _, _
694 if (!isScrolled()) // Buffer not scrolled, so we can simply reserve more space
695 {
696 _maxSize = targetSize + _Padding;
697 _data.reserve(_maxSize);
698 if (_dataEnd == 0) { _dataEnd = _data.size(); }
699 }
700 // se e s s e
701 // 6, 7, 3, 4, 5, // 5, 6, X, X, 2, 3, 4 // X, 6, 7, 8, 9, 10, X
702 // 6, 7, _, 3, 4, 5, // 5, 6, _, X, X, 2, 3, 4 // X, 6, 7, 8, 9, 10, _, _, X
703 else // (_dataStart != 0) // Buffer scrolled, so we need to copy the values to the correct positions
704 {
705 if (_dataEnd == 0) // Buffer actually not scrolled
706 {
707 _maxSize = targetSize + _Padding;
708 _data.reserve(_maxSize);
709 _dataEnd = _data.size();
710 return;
711 }
712
713 _data.resize(targetSize + _Padding);
714
715 auto copy_backward = [](auto first, auto last, auto d_last) { // Needed, as MacOS with C++23 does not support std::copy_backward
716 while (first != last)
717 {
718 *(--d_last) = *(--last);
719 }
720 };
721
722 copy_backward(std::next(_data.begin(), static_cast<int64_t>(_dataEnd)),
723 std::next(_data.begin(), static_cast<int64_t>(_maxSize)),
724 std::next(_data.begin(), static_cast<int64_t>(targetSize + _Padding)));
725
726 auto diff = targetSize + _Padding - _maxSize;
727 if (_dataStart >= _dataEnd)
728 {
729 _dataStart += diff;
730 }
731
732 _maxSize = targetSize + _Padding;
733 }
734 }
735 }
736 }
737

739 // Other
740 // ###########################################################################################################

742 /// @brief Returns the largest value in the buffer
743 [[nodiscard]] T max() const
744 {
745 T currentMax = front();
746 for (size_t i = 0; i < _data.size(); i++)
747 {
748 if (i >= _dataStart || i < _dataEnd)
749 {
750 currentMax = std::max(currentMax, _data.at(i));
751 }
752 }
753 return currentMax;
754 }

758 {

760 for (size_t i = 0; i < _data.size(); i++)
761 {
762 if (i >= _dataStart || i < _dataEnd)
763 {
764 currentMin = std::min(currentMin, _data.at(i));
765 }
766 }

773 return static_cast<int>(_dataStart);

794 /// @brief Converts the raw buffer to a string
795 [[nodiscard]] std::string getRawString() const
796 {
797 // Scrolled
798 // e s
799 // 5, 6, _, _, X, 2, 3, 4
800
801 // se
802 // 5, 6, 2, 3, 4
803
804 // s e
805 // _, 6, _, _, _
806
807 // Not scrolled
808 // s e
809 // X, X, 0, 1, 2, 3, _, _
810
811 // s e
812 // X, 6, 7, 8, 9, 10, _, _, X
813
814 // s e
815 // 6, X, X,
816
817 // s e
818 // 6, _, X, X,
819
820 // se
821 // X, X, _, _, _,
822
823 std::string out;
824
825 for (int i = 0; static_cast<size_t>(i) < _maxSize; i++) // X, 6, 7, 8, 9, 10, _, _, X
826 {
827 if ((i >= static_cast<int>(_dataStart - _Padding) && static_cast<size_t>(i) < _dataStart)
828 || (static_cast<size_t>(i) >= _dataStart + _maxSize - _Padding))
829 {
830 out += "X"; // padding
831 }
832 else if (bool scrolled = isScrolled();
833 static_cast<size_t>(i) >= _data.size()
834 || (scrolled && static_cast<size_t>(i) >= _dataEnd && (static_cast<int>(_dataStart - _Padding) < 0 || i < static_cast<int>(_dataStart - _Padding)))
835 || (scrolled && _dataStart < _dataEnd && (static_cast<size_t>(i) < _dataStart || static_cast<size_t>(i) >= _dataEnd))
836 || (_data.size() > 1 && !scrolled && static_cast<size_t>(i) >= _dataEnd))
837 {
838 out += "_"; // empty
839 }
840 else
841 {
842 out += std::to_string(_data.at(static_cast<size_t>(i)));
843 }
844 if (static_cast<size_t>(i) != _maxSize - 1)
845 {
846 out += ", ";
847 }
848 }
849 return out;
850 }

859 return os;

864 bool _infiniteBuffer{ false };
865 /// The maximum amount of objects to store in the buffer before overwriting itself when full
866 /// When _infiniteBuffer == true, then this corresponds to m_data.size()
867 size_t _maxSize;
868 /// The index of the first element in the scrolling buffer (0 if the buffer is empty)
869 size_t _dataStart{ 0 };
870 /// The index one after the last element (0 if the buffer is empty)
871 size_t _dataEnd{ 0 };
872 /// The data storage object
873 std::vector<T> _data;
874
875 /// @brief Checks if the buffer is scrolled
876 [[nodiscard]] bool isScrolled() const
877 {
878 if (_data.size() == 1) { return false; }
879
880 // e s
881 // 5, 6, _, _, X, 2, 3, 4
882
883 // se
884 if (_dataEnd == 0 && !empty()) // 1, 2, 3, 4
885 {
886 return true;
887 }
888
889 return _dataEnd < _dataStart
890 || (_dataStart != _Padding);
891
892 // se // se
893 // 5, 6, 2, 3, 4 // X, X, _, _, _
894
895 // s e
896 // X, 0, 1, 2, 3, _, _, X // Scrolled
897 // s e
898 // X, X, 0, 1, 2, 3, _, _ // Not scrolled
899 }
