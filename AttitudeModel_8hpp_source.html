33template<typename StateKeyType>
34class AttitudeModel
35{
36 public:
37 /// @brief Initializes the attitude model
38 void initialize()
39 {
40 for (int i = 0; i < _gui_covarianceAngularVelocity.rows(); i++)
41 {
42 _covarianceAngularVelocity(i) = convertUnit(_gui_covarianceAngularVelocity(i), _gui_covarianceAngularVelocityUnit);
43 }
44 }
45
46 /// @brief Updates the provided Phi, Q and G matrix
47 /// @param[in, out] Phi State transition matrix
48 /// @param[in, out] Q System/Process noise covariance matrix
49 /// @param[in] dt Time step size in [s]
50 /// @param[in] latitude Latitude [rad]
51 /// @param[in] longitude Longitude [rad]
52 /// @param[in] n_quat_b Quaternion body to navigation frame
53 template<typename Scalar, int Size, typename Derived>
54 void updatePhiAndQ(KeyedMatrix<Scalar, StateKeyType, StateKeyType, Size, Size>& Phi,
55 KeyedMatrix<Scalar, StateKeyType, StateKeyType, Size, Size>& Q,
56 double dt,
57 const double& latitude,
58 const double& longitude,
59 const Eigen::QuaternionBase<Derived>& n_quat_b)
60 {
61 Phi.template block<4>(Att, Att).setIdentity();
62 Q.template block<4>(Att, Att) = calcProcessNoiseMatrix(dt, latitude, longitude, n_quat_b)(all, all);
63 }
64
65 /// @brief Calculates the state transition matrix (𝚽) and the process noise covariance matrix (𝐐)
66 /// @param[in] dt Time step size in [s]
67 /// @param[in] latitude Latitude [rad]
68 /// @param[in] longitude Longitude [rad]
69 /// @param[in] n_quat_b Quaternion body to navigation frame
70 /// @return Phi and Q matrix
71 template<typename Derived>
72 [[nodiscard]] std::pair<KeyedMatrix4d<StateKeyType>, KeyedMatrix4d<StateKeyType>>
73 calcPhiAndQ(double dt, const double& latitude, const double& longitude, const Eigen::QuaternionBase<Derived>& n_quat_b)
74 {
75 KeyedMatrix4d<StateKeyType> Phi(Eigen::Matrix4d::Zero(), Att, Att);
76 KeyedMatrix4d<StateKeyType> Q(Eigen::Matrix4d::Zero(), Att, Att);
77 updatePhiAndQ(Phi, Q, dt, latitude, longitude, n_quat_b);
78
79 return { Phi, Q };
80 }
81
82 /// @brief Shows a GUI
83 /// @param[in] itemWidth Width of the space for the config items
84 /// @param[in] unitWidth Width of the units
85 /// @param[in] id Unique id for ImGui
86 /// @return True if something was changed
87 bool ShowGui(float itemWidth, float unitWidth, const char* id)
88 {
89 bool changed = false;
90
91 if (gui::widgets::InputDouble3LWithUnit(fmt::format("{} of the angular velocity due to user motion (Roll, Pitch, Yaw)##{}",
92 _gui_covarianceAngularVelocityUnit == Units::CovarianceAngularVelocityUnits::deg_sqrts
93 || _gui_covarianceAngularVelocityUnit == Units::CovarianceAngularVelocityUnits::degsqrts_min
94 || _gui_covarianceAngularVelocityUnit == Units::CovarianceAngularVelocityUnits::rad_sqrts
95 ? "StdDev"
96 : "Variance",
97 id)
98 .c_str(),
99 itemWidth, unitWidth, _gui_covarianceAngularVelocity.data(), 0.0, std::numeric_limits<double>::max(),
100 _gui_covarianceAngularVelocityUnit, MakeComboItems<Units::CovarianceAngularVelocityUnits>().c_str(),
101 "%.1e", ImGuiInputTextFlags_CharsScientific))
102 {
103 LOG_DEBUG("{}: _gui_covarianceAngularVelocity changed to {}", id, _gui_covarianceAngularVelocity.transpose());
104 LOG_DEBUG("{}: _gui_covarianceAngularVelocityUnit changed to {}", id, to_string(_gui_covarianceAngularVelocityUnit));
105 changed = true;
106 }
107
108 return changed;
109 }
110
111 private:
112 /// @brief All attitude keys
113 const std::array<StateKeyType, 4>& Att = Keys::Att<StateKeyType>;
114
115 /// @brief Calculates the process noise matrix Q
116 /// @param[in] dt Time step [s]
117 /// @param[in] latitude Latitude [rad]
118 /// @param[in] longitude Longitude [rad]
119 /// @param[in] n_quat_b Quaternion body to navigation frame
120 template<typename Derived>
121 [[nodiscard]] KeyedMatrix4d<StateKeyType>
122 calcProcessNoiseMatrix(double dt, double latitude, double longitude, const Eigen::QuaternionBase<Derived>& n_quat_b) const
123 {
124 // Scaling matrix in n-frame
125 Eigen::Matrix3d S_n = _covarianceAngularVelocity.asDiagonal();
126
127 Eigen::Matrix4d n_covQuat_b = trafo::covRPY2quat(S_n, n_quat_b);
128 Eigen::Matrix4d e_J_n = trafo::covQuat2QuatJacobian(trafo::e_Quat_n(latitude, longitude));
129
130 // Scaling matrix in e-frame
131 Eigen::Matrix4d S_e = e_J_n * n_covQuat_b * e_J_n.transpose();
132
133 return { dt * S_e, Att, Att };
134 }
135
136 /// Gui selection for the Unit of the input covarianceAngularVelocity
137 Units::CovarianceAngularVelocityUnits _gui_covarianceAngularVelocityUnit = Units::CovarianceAngularVelocityUnits::deg_sqrts;
138 /// @brief GUI selection for the Standard deviation of the angular velocity due to user motion (Roll, Pitch, Yaw)
139 Eigen::Vector3d _gui_covarianceAngularVelocity = { 100.0, 100.0, 100.0 } /* [ deg / √(s) ] */;
140
141 /// @brief Covariance of the angular velocity due to user motion (Roll, Pitch, Yaw) [rad²/s]
142 Eigen::Vector3d _covarianceAngularVelocity;
143

145 /// @param[out] j Json object which gets filled with the info

147 friend void to_json(json& j, const AttitudeModel& data)

156 /// @param[out] data Object to fill from the json

160 if (j.contains("covarianceAngularVelocity")) { j.at("covarianceAngularVelocity").get_to(data._gui_covarianceAngularVelocity); }
161 }
162};
