216std::pair<uint8_t, uint8_t> NAV::vendor::emlid::checksumUBX(const std::vector<uint8_t>& data)
217{
218 uint8_t cka = 0;
219 uint8_t ckb = 0;
220
221 for (size_t i = 2; i < data.size() - 2; i++)
222 {
223 cka += data.at(i);
224 ckb += cka;
225 }
226 return std::make_pair(cka, ckb);
227}
